## 一、SaaS 落地页 SEO 架构

### 1.1 页面结构规划

```
SaaS Landing Page (CloudBoard - 团队协作看板)
│
├── <head> — SEO 元数据 + 结构化数据
│   ├── Title (55 字符)
│   ├── Meta Description (155 字符)
│   ├── Canonical + Hreflang
│   ├── Open Graph + Twitter Card
│   ├── JSON-LD: SoftwareApplication
│   ├── JSON-LD: FAQPage (内联)
│   └── Preload 关键资源 (字体/首屏图片)
│
├── <body>
│   ├── Skip Link (无障碍)
│   │
│   ├── <header> — 导航
│   │   ├── <nav aria-label="主导航">
│   │   ├── Logo (h1 或 img alt)
│   │   ├── 导航链接
│   │   └── CTA 按钮
│   │
│   ├── <main>
│   │   ├── <section id="hero"> — Hero 区 (h1)
│   │   ├── <section id="features"> — 功能特性 (h2)
│   │   │   └── <article> × 4 (功能卡片)
│   │   ├── <section id="pricing"> — 定价 (h2)
│   │   │   └── <article> × 3 (定价卡片)
│   │   ├── <section id="testimonials"> — 用户评价 (h2)
│   │   │   └── <blockquote> × 3
│   │   ├── <section id="faq"> — FAQ (h2)
│   │   │   └── <details> × 5
│   │   └── <section id="cta"> — 最终 CTA (h2)
│   │
│   └── <footer>
│       ├── 链接分组 (<nav>)
│       ├── 版权信息
│       └── 联系方式
```

### 1.2 关键词策略

```html
    // ===== 1. 模态框焦点管理 =====
    const modal = document.getElementById('signup-modal');
    const openBtn = document.getElementById('open-signup-modal');
    const closeBtn = document.getElementById('close-modal');
    let lastFocusedElement = null;

    function openModal() {
      lastFocusedElement = document.activeElement;
      modal.classList.add('active');
      modal.setAttribute('aria-hidden', 'false');
      document.body.style.overflow = 'hidden';
      // 聚焦到关闭按钮
      closeBtn.focus();
      // 阻止背景内容被 tab 访问
      trapFocus(modal.querySelector('.modal'));
    }

    function closeModal() {
      modal.classList.remove('active');
      modal.setAttribute('aria-hidden', 'true');
      document.body.style.overflow = '';
      // 恢复焦点
      if (lastFocusedElement) lastFocusedElement.focus();
    }

    openBtn.addEventListener('click', openModal);
    closeBtn.addEventListener('click', closeModal);

    // 点击遮罩关闭
    modal.addEventListener('click', (e) => {
      if (e.target === modal) closeModal();
    });

    // Escape 键关闭
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && modal.classList.contains('active')) {
        closeModal();
      }
    });

    // 焦点陷阱 (Focus Trap)
    function trapFocus(container) {
      const focusable = container.querySelectorAll(
        'a[href], button:not([disabled]), input, textarea, select, [tabindex]:not([tabindex="-1"])'
      );
      const first = focusable[0];
      const last = focusable[focusable.length - 1];

      container.addEventListener('keydown', function handler(e) {
        if (e.key !== 'Tab') return;
        if (e.shiftKey) {
          if (document.activeElement === first) {
            e.preventDefault();
            last.focus();
          }
        } else {
          if (document.activeElement === last) {
            e.preventDefault();
            first.focus();
          }
        }
      });
    }

    // ===== 2. 表单验证 + 无障碍错误提示 =====

      let isValid = true;

      // 姓名验证
      const nameInput = document.getElementById('signup-name');
      const nameGroup = document.getElementById('name-group');
      const nameError = document.getElementById('name-error');
      if (!nameInput.value.trim()) {
        nameGroup.classList.add('has-error');
        nameError.textContent = '请输入你的姓名';
        nameInput.setAttribute('aria-invalid', 'true');
        isValid = false;
      } else {
        nameGroup.classList.remove('has-error');
        nameError.textContent = '';
        nameInput.removeAttribute('aria-invalid');
      }

      // 邮箱验证
      const emailInput = document.getElementById('signup-email');
      const emailGroup = document.getElementById('email-group');
      const emailError = document.getElementById('email-error');
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(emailInput.value)) {
        emailGroup.classList.add('has-error');
        emailError.textContent = '请输入有效的邮箱地址';
        emailInput.setAttribute('aria-invalid', 'true');
        isValid = false;
      } else {
        emailGroup.classList.remove('has-error');
        emailError.textContent = '';
        emailInput.removeAttribute('aria-invalid');
      }

      // 密码验证
      const passwordInput = document.getElementById('signup-password');
      const passwordGroup = document.getElementById('password-group');
      const passwordError = document.getElementById('password-error');
      if (passwordInput.value.length < 8) {
        passwordGroup.classList.add('has-error');
        passwordError.textContent = '密码至少需要 8 位字符';
        passwordInput.setAttribute('aria-invalid', 'true');
        isValid = false;
      } else {
        passwordGroup.classList.remove('has-error');
        passwordError.textContent = '';
        passwordInput.removeAttribute('aria-invalid');
      }

      if (!isValid) {
        // 聚焦到第一个错误字段
        const firstError = form.querySelector('[aria-invalid="true"]');
        if (firstError) firstError.focus();
        // 显示 Toast 通知
        showToast('请修正表单中的错误', 'error');
        return;
      }

      // 成功
      closeModal();
      showToast('账户创建成功！欢迎使用 CloudBoard 🎉', 'success');
      form.reset();
    });

    // 输入时清除错误状态
    form.querySelectorAll('input').forEach(input => {
      input.addEventListener('input', () => {
        const group = input.closest('.form-group');
        if (group) group.classList.remove('has-error');
        input.removeAttribute('aria-invalid');
        const errorEl = input.closest('.form-group')?.querySelector('.error-msg');
        if (errorEl) errorEl.textContent = '';
      });
    });

    // ===== 3. Toast 通知 =====
    function showToast(message, type = 'info') {
      const container = document.querySelector('.toast-container');
      const toast = document.createElement('div');
      toast.className = 'toast';
      toast.setAttribute('role', 'status');
      toast.textContent = message;
      if (type === 'error') toast.style.borderLeft = '4px solid #e63946';
      if (type === 'success') toast.style.borderLeft = '4px solid #06d6a0';
      container.appendChild(toast);

      // 3 秒后自动移除
      setTimeout(() => {
        toast.style.opacity = '0';
        toast.style.transform = 'translateX(100%)';
        toast.style.transition = 'all 0.3s';
        setTimeout(() => toast.remove(), 300);
      }, 3000);
    }

    // ===== 4. 平滑滚动 + 焦点管理 =====
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
      anchor.addEventListener('click', (e) => {
        const target = document.querySelector(anchor.getAttribute('href'));
        if (target) {
          e.preventDefault();
          target.scrollIntoView({ behavior: 'smooth', block: 'start' });
          // 滚动后设置焦点，方便屏幕阅读器定位
          target.setAttribute('tabindex', '-1');
          target.focus({ preventScroll: true });
        }
      });
    });

    // ===== 5. FAQ <details> 键盘增强 =====
    document.querySelectorAll('.faq-item').forEach(item => {
      const summary = item.querySelector('summary');
      summary.addEventListener('keydown', (e) => {
        // Enter/Space 已原生支持，这里处理箭头键导航
        if (e.key === 'ArrowDown' && item.nextElementSibling?.classList.contains('faq-item')) {
          e.preventDefault();
          item.nextElementSibling.querySelector('summary').focus();
        }
        if (e.key === 'ArrowUp' && item.previousElementSibling?.classList.contains('faq-item')) {
          e.preventDefault();
          item.previousElementSibling.querySelector('summary').focus();
        }
      });
    });
  </script>
</body>
</html>
```

## 三、核心优化点详解

### 3.1 语义化 HTML 关键点

| 优化 | Before (反模式) | After (正确) | 影响 |
|------|----------------|--------------|------|
| 页面分区 | `<div class="header">` | `<header role="banner">` | SEO + A11y |
| 导航 | `<div class="nav">` | `<nav aria-label="主导航">` | A11y 导航 |
| 主内容 | `<div class="content">` | `<main id="main-content">` | SEO + A11y |
| 功能卡片 | `<div class="card">` | `<article class="feature-card">` | 语义独立性 |
| 评价 | `<div class="review">` | `<blockquote> + <cite>` | 语义引用 |
| FAQ | `<div>` + JS toggle | `<details> + <summary>` | 原生可访问 |
| 定价卡片 | `<div>` + JS | `<article role="list">` | 语义 + 列表关系 |
| 页脚 | `<div class="footer">` | `<footer role="contentinfo">` | SEO + A11y |
| 链接文本 | `<a href="#">点击这里</a>` | `<a href="#features">了解更多 ↓</a>` | SEO 锚文本 + A11y |
| 图片 | `<img src="pic.jpg">` | `<img alt="描述性文本" width height>` | SEO 图片搜索 + A11y |

### 3.2 ARIA 使用清单

| ARIA 属性/角色 | 使用位置 | 作用 |
|----------------|----------|------|
| `aria-label="主导航"` | `<nav>` | 区分多个导航区域 |
| `aria-label="产品导航"` | 页脚 `<nav>` | 区分页脚内多个导航 |
| `aria-labelledby="hero-heading"` | `<section>` | 区域与标题关联 |
| `aria-label="评分 5 星"` | `.stars` | 图标转可读文本 |
| `role="img"` | `.stars` | 声明图标语义 |
| `aria-hidden="true"` | 装饰图标 | 屏幕阅读器跳过 |
| `aria-modal="true"` | 模态框 | 声明模态对话框 |
| `aria-hidden="true/false"` | 模态框 | 动态控制可见性 |
| `aria-describedby` | 表单输入 | 关联帮助/错误文本 |
| `aria-required="true"` | 必填输入 | 声明必填状态 |
| `aria-invalid="true"` | 错误输入 | 声明验证失败 |
| `role="alert"` + `aria-live="polite"` | 错误消息 | 实时通知屏幕阅读器 |
| `role="status"` + `aria-live="polite"` | Toast 容器 | 非侵入式通知 |
| `aria-label="返回首页"` | Logo 链接 | 描述链接目的 |
| `aria-label="关闭注册对话框"` | 关闭按钮 | 描述按钮功能 |
| `skip-link` | 页面顶部 | 键盘用户跳过导航 |
| `:focus-visible` | 全局样式 | 键盘焦点可见 |
| `prefers-reduced-motion` | CSS 媒体查询 | 尊重动画偏好 |
| `prefers-contrast: high` | CSS 媒体查询 | 高对比度模式 |

### 3.3 结构化数据清单

| Schema 类型 | 作用 | Rich Snippet 效果 |
|-------------|------|-------------------|
| `SoftwareApplication` | 应用信息 | 评分星、价格、应用类别 |
| `FAQPage` | FAQ 内容 | 搜索结果直接展示问答 |
| `BreadcrumbList` | 页面层级 | 搜索结果面包屑展示 |
| `AggregateRating` | 评分汇总 | 搜索结果星等显示 |

### 3.4 Core Web Vitals SEO 优化

| 指标 | 目标 | 优化措施 |
|------|------|----------|
| **LCP** (最大内容绘制) | < 2.5s | `preload` 首屏图片 + `fetchpriority="high"` + WebP 格式 + 明确 width/height |
| **INP** (交互延迟) | < 200ms | CSS 动画用 `transform/opacity` + 避免主线程长任务 + 事件委托 |
| **CLS** (累积布局偏移) | < 0.1 | 图片明确 width/height + 字体 `preload` + 广告/动态内容预留空间 |

### 3.5 无障碍审计流程

## 六、关键知识点速查

### 6.1 语义标签选择速查

```
独立内容 → <article>
导航区域 → <nav aria-label="xxx">
内容分区 → <section aria-labelledby="heading-id">
  └─ 无标题纯布局 → <div> (不要用 section!)
辅助内容 → <aside>
页眉/页脚 → <header> / <footer>
可展开内容 → <details> + <summary> (原生手风琴)
引用 → <blockquote> / <q>
时间 → <time datetime="ISO-8601">
定义 → <dfn>
代码 → <code> / <pre>
列表 → <ul> / <ol> / <dl>
```

### 6.2 ARIA 速查

```
角色: role="dialog" | "alert" | "status" | "navigation" | "main" | "contentinfo"
状态: aria-hidden | aria-expanded | aria-selected | aria-checked | aria-disabled
关系: aria-labelledby | aria-describedby | aria-controls | aria-owns
实时: aria-live="polite" | "assertive" | "off"
表单: aria-required | aria-invalid | aria-errormessage
```

### 6.3 JSON-LD 速查

```
常用类型: WebPage / Article / Product / FAQPage / HowTo
        / SoftwareApplication / LocalBusiness / Event
```
