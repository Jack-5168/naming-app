## 章节 1：闭包 (Closures)

### 核心概念

闭包是指函数能够记住并访问它的词法作用域，即使函数在其词法作用域之外执行。

### 关键知识点

#### 1.1 闭包的形成

```javascript
function makeCounter() {
  let count = 0;

  return function () {
    return count++;
  };
}

const counter = makeCounter();
console.log(counter()); // 0
console.log(counter()); // 1
console.log(counter()); // 2
```

#### 1.2 闭包的本质

- 函数 + 词法环境的引用 = 闭包
- 闭包让内部函数可以访问外部函数的变量
- 外部函数执行完毕后，变量不会被垃圾回收

#### 1.3 实际应用

```javascript
// 模块化模式
function createModule() {
  let privateVar = "secret";

  return {
    getPrivate: function () {
      return privateVar;
    },
    setPrivate: function (val) {
      privateVar = val;
    },
  };
}

// 函数工厂
function createMultiplier(multiplier) {
  return function (num) {
    return num * multiplier;
  };
}

const double = createMultiplier(2);
const triple = createMultiplier(3);

console.log(double(5)); // 10
console.log(triple(5)); // 15
```

#### 1.4 常见陷阱

```javascript
// 错误示例：循环中的闭包
for (var i = 0; i < 3; i++) {
  setTimeout(function () {
    console.log(i); // 3, 3, 3
  }, 100);
}

// 正确示例 1：使用 let
for (let i = 0; i < 3; i++) {
  setTimeout(function () {
    console.log(i); // 0, 1, 2
  }, 100);
}

// 正确示例 2：使用 IIFE
for (var i = 0; i < 3; i++) {
  (function (j) {
    setTimeout(function () {
      console.log(j); // 0, 1, 2
    }, 100);
  })(i);
}
```

## 章节 2：原型与继承 (Prototypes and Inheritance)

### 核心概念

JavaScript 使用原型链实现继承，每个对象都有一个指向其原型的内部链接。

### 关键知识点

#### 2.1 原型链

```javascript
const animal = {
  eats: true,
  walk() {
    console.log("Animal walks");
  },
};

const rabbit = {
  jumps: true,
  __proto__: animal,
};

console.log(rabbit.eats); // true (从原型继承)
rabbit.walk(); // Animal walks (从原型继承)
```

#### 2.2 constructor 和 prototype

```javascript
function Person(name) {
  this.name = name;
}

Person.prototype.sayHello = function () {
  console.log(`Hello, I'm ${this.name}`);
};

const person1 = new Person("Alice");
person1.sayHello(); // Hello, I'm Alice

console.log(Person.prototype); // { sayHello: [Function] }
console.log(person1.__proto__ === Person.prototype); // true
```

#### 2.3 class 语法糖

```javascript
class Animal {
  constructor(name) {
    this.name = name;
  }

  walk() {
    console.log(`${this.name} walks`);
  }
}

class Rabbit extends Animal {
  constructor(name, speed) {
    super(name);
    this.speed = speed;
  }

  run() {
    console.log(`${this.name} runs at ${this.speed}`);
  }

  // 重写父类方法
  walk() {
    console.log(`${this.name} hops`);
  }
}

const rabbit = new Rabbit("White Rabbit", "fast");
rabbit.walk(); // White Rabbit hops
rabbit.run(); // White Rabbit runs at fast
```

#### 2.4 原型链查找

```javascript
const obj = {
  a: 1,
};

const proto1 = {
  b: 2,
  __proto__: obj,
};

const proto2 = {
  c: 3,
  __proto__: proto1,
};

console.log(proto2.a); // 1 (沿着原型链查找)
console.log(proto2.b); // 2
console.log(proto2.c); // 3
console.log(proto2.toString); // [Function: toString] (Object.prototype)
```

## 章节 3：异步编程 (Async Programming)

### 核心概念

JavaScript 是单线程的，通过事件循环和异步机制处理耗时操作。

### 关键知识点

#### 3.1 Promise 基础

```javascript
const promise = new Promise((resolve, reject) => {
  setTimeout(() => {
    resolve("Success!");
  }, 1000);
});

promise
  .then((result) => console.log(result))
  .catch((error) => console.error(error))
  .finally(() => console.log("Done"));
```

#### 3.2 Promise 链式调用

```javascript
fetch("/api/user")
  .then((response) => response.json())
  .then((user) => fetch(`/api/posts/${user.id}`))
  .then((response) => response.json())
  .then((posts) => console.log(posts))
  .catch((error) => console.error("Error:", error));
```

#### 3.3 async/await

```javascript
async function getUserPosts(userId) {
  try {
    const userResponse = await fetch("/api/user");
    const user = await userResponse.json();

    const postsResponse = await fetch(`/api/posts/${user.id}`);
    const posts = await postsResponse.json();

    return posts;
  } catch (error) {
    console.error("Error:", error);
    throw error;
  }
}

// 并行执行
async function getData() {
  const [users, posts, comments] = await Promise.all([
    fetch("/api/users").then((r) => r.json()),
    fetch("/api/posts").then((r) => r.json()),
    fetch("/api/comments").then((r) => r.json()),
  ]);

  return { users, posts, comments };
}
```

#### 3.4 微任务与宏任务

```javascript
console.log("Script start");

setTimeout(() => {
  console.log("setTimeout"); // 宏任务
}, 0);

Promise.resolve()
  .then(() => {
    console.log("Promise 1"); // 微任务
  })
  .then(() => {
    console.log("Promise 2"); // 微任务
  });

console.log("Script end");

// 输出顺序:
// Script start
// Script end
// Promise 1
// Promise 2
// setTimeout
```

## 章节 4：事件循环 (Event Loop)

### 核心概念

事件循环是 JavaScript 处理异步操作的核心机制。

### 执行顺序

1. 执行同步代码（调用栈）
2. 执行所有微任务（Promise.then、MutationObserver 等）
3. 渲染 DOM（如果需要）
4. 执行一个宏任务（setTimeout、setInterval、I/O 等）
5. 重复步骤 2-4

### 示例分析

```javascript
console.log("1");

setTimeout(() => {
  console.log("2");
  Promise.resolve().then(() => {
    console.log("3");
  });
}, 0);

Promise.resolve().then(() => {
  console.log("4");
});

console.log("5");

// 输出顺序: 1, 5, 4, 2, 3
```

### 实战应用

```javascript
// 批量处理任务，避免阻塞 UI
async function processQueue(queue) {
  for (const item of queue) {
    await processItem(item);
    // 让出主线程，允许 UI 更新
    await new Promise((resolve) => setTimeout(resolve, 0));
  }
}

// 防抖函数
function debounce(fn, delay) {
  let timer = null;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => {
      fn.apply(this, args);
    }, delay);
  };
}

// 节流函数
function throttle(fn, limit) {
  let inThrottle = false;
  return function (...args) {
    if (!inThrottle) {
      fn.apply(this, args);
      inThrottle = true;
      setTimeout(() => (inThrottle = false), limit);
    }
  };
}
```
