## 一、闭包深度解析

### 1.1 词法环境的内部机制

JS 引擎中，每个执行上下文有一个 **LexicalEnvironment** 对象：

```
LexicalEnvironment = {
  EnvironmentRecord: {  // 变量绑定
    <identifier>: <value>,
    // ...
  },
  outer: <LexicalEnvironment reference>  // 指向外层环境
}
```

**关键理解：** 闭包不是「函数记住了变量」，而是「函数持有一个指向其创建时词法环境的引用」。

```javascript
function outer() {
  let a = 1;
  let b = 2;
  
  function inner() {
    return a;  // 只用了 a
  }
  
  return inner;
}

const fn = outer();
```

**引擎优化（V8 的闭包裁剪）：** 理论上 inner 只捕获了 `a`，但实际 V8 会捕获整个环境记录（因为 `a` 和 `b` 在同一个 EnvironmentRecord 中）。这意味着 `b` 也不会被 GC。

```javascript
// 验证闭包不释放外部变量
function createHugeClosure() {
  const hugeArray = new Array(10_000_000).fill('x');  // ~80MB
  
  return function() {
    return 42;  // 根本没用到 hugeArray！
  };
}

const fn = createHugeClosure();
// hugeArray 不会被 GC，因为闭包引用了整个词法环境
```

### 1.2 闭包的实战高级模式

#### 模式 A：惰性初始化（Lazy Initialization）

```javascript
function lazyInit(createValue) {
  let cached = undefined;
  let initialized = false;
  
  return function() {
    if (!initialized) {
      cached = createValue();
      initialized = true;
    }
    return cached;
  };
}

// 使用：数据库连接只创建一次
const getDB = lazyInit(() => {
  console.log('Connecting to DB...');
  return { connected: true };
});

getDB();  // Connecting to DB... → { connected: true }
getDB();  // { connected: true }（直接返回缓存）
```

#### 模式 B：状态机（State Machine）

```javascript
function createStateMachine(initial, transitions) {
  let state = initial;
  
  return {
    getState: () => state,
    transition(event) {
      const next = transitions[state]?.[event];
      if (!next) throw new Error(`Invalid transition: ${state} + ${event}`);
      state = next;
      return state;
    }
  };
}

const light = createStateMachine('red', {
  red:    { timer: 'green' },
  green:  { timer: 'yellow' },
  yellow: { timer: 'red' }
});

light.transition('timer');  // 'green'
light.transition('timer');  // 'yellow'
light.transition('timer');  // 'red'
```

#### 模式 C：柯里化 + 闭包

```javascript
function curry(fn, arity = fn.length) {
  return function curried(...args) {
    if (args.length >= arity) {
      return fn.apply(this, args);
    }
    return curried.bind(null, ...args);
  };
}

const add = (a, b, c) => a + b + c;
const curriedAdd = curry(add);

curriedAdd(1)(2)(3);       // 6
curriedAdd(1, 2)(3);       // 6
curriedAdd(1)(2, 3);       // 6
```

### 1.3 闭包经典陷阱 — 面试高频

#### 陷阱 1：循环变量捕获

```javascript
// ❌ 经典错误
const funcs = [];
for (var i = 0; i < 3; i++) {
  funcs.push(() => console.log(i));
}
funcs.forEach(f => f());  // 3, 3, 3

// ✅ 方案 1：let 块级作用域
for (let i = 0; i < 3; i++) {
  funcs.push(() => console.log(i));
}
// 0, 1, 2 — 每次迭代创建新的词法环境

// ✅ 方案 2：函数参数创建新绑定
for (var i = 0; i < 3; i++) {
  (function(j) {
    funcs.push(() => console.log(j));
  })(i);
}
```

#### 陷阱 2：闭包中的 `this`

```javascript
const obj = {
  name: 'Alice',
  methods: ['a', 'b', 'c'],
  
  printAll() {
    // ❌ 错误：箭头函数捕获外层 this（正确），但 forEach 的回调不是
    this.methods.forEach(function(method) {
      console.log(this.name, method);  // this 指向 undefined（严格模式）
    });
    
    // ✅ 正确：箭头函数
    this.methods.forEach(method => {
      console.log(this.name, method);  // Alice a, Alice b, Alice c
    });
    
    // ✅ 正确：bind
    this.methods.forEach(function(method) {
      console.log(this.name, method);
    }.bind(this));
  }
};
```

#### 陷阱 3：内存泄漏

```javascript
// ❌ 闭包持有 DOM 引用导致泄漏
function setup() {
  const element = document.getElementById('btn');
  
  element.addEventListener('click', function handler() {
    console.log('clicked');
    // handler 闭包持有了 element 的引用
    // 如果 element 被从 DOM 移除，但监听器没 remove，element 无法 GC
  });
  
  // element = null;  // 即使置 null，监听器闭包仍持有引用
}

// ✅ 正确做法
function setup() {
  const element = document.getElementById('btn');
  
  function handler() {
    console.log('clicked');
  }
  
  element.addEventListener('click', handler);
  
  // 清理时
  element.removeEventListener('click', handler);
}
```

## 二、原型与继承深度解析

### 2.1 原型链的完整结构

```
实例对象
  ↓ [[Prototype]] (Object.getPrototypeOf)
构造函数.prototype
  ↓ [[Prototype]]
父构造函数.prototype
  ↓ [[Prototype]]
Object.prototype
  ↓ [[Prototype]]
null
```

**关键区别：**
- `obj.__proto__` / `Object.getPrototypeOf(obj)` → 对象的原型
- `Constructor.prototype` → 构造函数的原型（给实例用的）
- `Constructor.__proto__` → 构造函数本身的原型（指向 Function.prototype）

```javascript
function Person(name) {
  this.name = name;
}

// Person 是函数 → 它的原型是 Function.prototype
Person.__proto__ === Function.prototype;  // true

// Person.prototype 是给 new Person() 的实例用的
const p = new Person('Alice');
p.__proto__ === Person.prototype;  // true

// Person.prototype 本身是对象 → 它的原型是 Object.prototype
Person.prototype.__proto__ === Object.prototype;  // true

// 完整链条
p.__proto__.__proto__ === Object.prototype;  // true
```

### 2.2 class 的本质（语法糖拆解）

```javascript
// 写出来的代码
class Animal {
  constructor(name) {
    this.name = name;
  }
  speak() { console.log(`${this.name} speaks`); }
  static create(name) { return new Animal(name); }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);
    this.breed = breed;
  }
  speak() { console.log(`${this.name}: woof!`); }
}

// === 等价的原型写法 ===
function Animal(name) {
  this.name = name;
}
Animal.prototype.speak = function() { console.log(`${this.name} speaks`); };
Animal.create = function(name) { return new Animal(name); };  // 静态方法

function Dog(name, breed) {
  Animal.call(this, name);  // super(name)
  this.breed = breed;
}

// 继承原型链
Dog.prototype = Object.create(Animal.prototype);
Dog.prototype.constructor = Dog;

// 继承静态方法
Object.setPrototypeOf(Dog, Animal);

Dog.prototype.speak = function() { console.log(`${this.name}: woof!`); };
```

### 2.3 六种继承模式对比

```javascript
// 1. 原型链继承
function Parent1() { this.names = ['a']; }
function Child1() {}
Child1.prototype = new Parent1();
// ❌ 问题：引用类型共享，子类实例互相影响

// 2. 构造函数继承（借用构造函数）
function Parent2() { this.names = ['a']; }
function Child2() { Parent2.call(this); }
// ❌ 问题：无法继承原型方法，每次创建都调用父构造函数

// 3. 组合继承（最常用）
function Parent3() { this.names = ['a']; }
Parent3.prototype.say = function() {};
function Child3() { Parent3.call(this); }  // 第二次调用
Child3.prototype = new Parent3();  // 第一次调用
Child3.prototype.constructor = Child3;
// ⚠️ 问题：父构造函数被调用两次

// 4. 原型式继承
function object(o) {
  function F() {}
  F.prototype = o;
  return new F();
}
const parent = { names: ['a'] };
const child = object(parent);
// 适合不需要构造函数的场景

// 5. 寄生式继承
function createAnother(original) {
  const clone = object(original);
  clone.sayHi = function() { console.log('hi'); };
  return clone;
}

// 6. 寄生组合继承（最优）⭐
function inherit(Child, Parent) {
  const prototype = Object.create(Parent.prototype);
  prototype.constructor = Child;
  Child.prototype = prototype;
}

function Parent6(name) { this.name = name; }
Parent6.prototype.say = function() {};

function Child6(name, age) {
  Parent6.call(this, name);  // 只调用一次
  this.age = age;
}
inherit(Child6, Parent6);

const c = new Child6('Alice', 20);
```

### 2.4 原型链实战 — 实现 instanceof

```javascript
function myInstanceof(obj, Constructor) {
  let proto = Object.getPrototypeOf(obj);
  const prototype = Constructor.prototype;
  
  while (proto !== null) {
    if (proto === prototype) return true;
    proto = Object.getPrototypeOf(proto);
  }
  return false;
}

// 测试
myInstanceof([], Array);        // true
myInstanceof([], Object);       // true
myInstanceof({}, Array);        // false
myInstanceof(null, Object);     // false
```

## 三、异步编程深度解析

### 3.1 Promise 状态机

```
         pending
          /    \
     resolve   reject
        /        \
   fulfilled    rejected
      |             |
    .then        .catch
      \           /
       \         /
      settled (终态，不可变)
```

**关键特性：**
1. 状态一旦改变就不可逆（fulfilled → 不能再变 rejected）
2. `.then` 总是返回 **新的 Promise**（不是原 Promise）
3. `.then` 的回调返回值决定新 Promise 的状态

```javascript
// 链式调用的本质：每次 .then 返回新 Promise
const p1 = new Promise(r => r(1));
const p2 = p1.then(v => v + 1);  // p2 是新的 Promise，值为 2
const p3 = p2.then(v => v * 2);  // p3 是新的 Promise，值为 4

p1 === p2;  // false
p2 === p3;  // false
```

### 3.2 Promise 实现（简化版）

```javascript
class MyPromise {
  constructor(executor) {
    this.state = 'pending';
    this.value = undefined;
    this.callbacks = [];
    
    const resolve = (value) => {
      if (this.state !== 'pending') return;
      this.state = 'fulfilled';
      this.value = value;
      this.callbacks.forEach(cb => cb.onFulfilled?.(value));
    };
    
    const reject = (reason) => {
      if (this.state !== 'pending') return;
      this.state = 'rejected';
      this.value = reason;
      this.callbacks.forEach(cb => cb.onRejected?.(reason));
    };
    
    try {
      executor(resolve, reject);
    } catch (e) {
      reject(e);
    }
  }
  
  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      const handle = () => {
        try {
          const callback = this.state === 'fulfilled' ? onFulfilled : onRejected;
          if (typeof callback !== 'function') {
            this.state === 'fulfilled' ? resolve(this.value) : reject(this.value);
            return;
          }
          const result = callback(this.value);
          if (result instanceof MyPromise) {
            result.then(resolve, reject);
          } else {
            resolve(result);
          }
        } catch (e) {
          reject(e);
        }
      };
      
      if (this.state === 'pending') {
        this.callbacks.push({ onFulfilled: handle, onRejected: handle });
      } else {
        // 异步执行（微任务）
        queueMicrotask(handle);
      }
    });
  }
  
  catch(onRejected) {
    return this.then(null, onRejected);
  }
}
```

### 3.3 async/await 的本质

```javascript
// async 函数 = 返回 Promise 的函数
async function foo() {
  return 42;
}
foo();  // Promise { <fulfilled>: 42 }

// await = 暂停执行，等待 Promise 解决
async function bar() {

  const y = await Promise.resolve(2);

// === 等价于 ===

  );
}
```

**await 的陷阱：**

## 四、事件循环深度解析

### 4.1 浏览器事件循环完整模型

```
┌─────────────────────────────────────┐
│           Call Stack                │
│  (同步代码执行)                      │
└──────────────┬──────────────────────┘
               │
    ┌──────────┴──────────┐
    │                     │
    ▼                     ▼
┌─────────┐         ┌──────────┐
│ Microtask│         │ Macrotask│
│  Queue   │         │  Queue   │
│          │         │          │
│ • Promise│         │ • setTimeout│
│ • queueMicrotask│   │ • setInterval│
│ • MutationObserver│ │ • I/O       │
│ • process.nextTick│ │ • UI rendering│
│   (Node.js)       │ │ • message    │
└─────────┘         └──────────┘
    │                     │
    └──────┬──────────────┘
           │
           ▼
    ┌─────────────┐
    │  Render     │
    │  (如果需要)  │
    └─────────────┘
```

**执行规则：**
1. 执行完当前调用栈的所有同步代码
2. **清空** 微任务队列（全部执行完，不是一次）
3. 执行一次渲染（如果页面需要更新）
4. 从宏任务队列取 **一个** 任务执行
5. 回到步骤 2

### 4.2 经典面试题

```javascript
console.log('1');           // 同步

setTimeout(() => {          // 宏任务 1
  console.log('2');
  Promise.resolve().then(() => {  // 微任务
    console.log('3');
  });
}, 0);

Promise.resolve().then(() => {  // 微任务 1
  console.log('4');
  Promise.resolve().then(() => {  // 微任务 2
    console.log('5');
  });
});

console.log('6');           // 同步

// 输出: 1, 6, 4, 5, 2, 3
//
// 执行流程:
// 1. 同步: 1, 6
// 2. 微任务队列: [then1] → 执行 then1 → 输出 4, 注册 then2
// 3. 微任务队列: [then2] → 执行 then2 → 输出 5
// 4. 渲染
// 5. 宏任务: setTimeout → 输出 2, 注册微任务
// 6. 微任务: 输出 3
```

### 4.3 Node.js 事件循环（与浏览器的差异）

```
Node.js 事件循环阶段:

  ┌───────────────────────────┐
│ 1. timers          │  setTimeout/setInterval 回调
│ 2. pending callbacks │  系统操作回调
│ 3. idle, prepare    │  内部使用
│ 4. poll             │  I/O 回调，最重要的阶段
│ 5. check            │  setImmediate 回调
│ 6. close callbacks  │  关闭回调 (socket.on('close'))
  └───────────────────────────┘

每个阶段之间有微任务队列（Promise.then, queueMicrotask）
process.nextTick 有独立的更高优先级队列
```

```javascript
// Node.js 特有行为
setTimeout(() => console.log('timeout'), 0);
setImmediate(() => console.log('immediate'));
// 输出顺序不确定！取决于事件循环进入哪个阶段

// 但如果有 I/O 操作：
fs.readFile('file.txt', () => {
  setTimeout(() => console.log('timeout'), 0);
  setImmediate(() => console.log('immediate'));
});
// 输出: immediate, timeout（setImmediate 总是在 poll 之后执行）
```

```javascript
// process.nextTick 优先级最高
Promise.resolve().then(() => console.log('promise'));
process.nextTick(() => console.log('nextTick'));
console.log('sync');

// Node.js 输出: sync, nextTick, promise
// 浏览器不支持 process.nextTick
```

### 4.5 性能优化：避免事件循环阻塞

```javascript
// ❌ 阻塞主线程
function processLargeArray(arr) {
  for (let i = 0; i < arr.length; i++) {
    // 处理每个元素 → 可能阻塞几百毫秒
    processItem(arr[i]);
  }
}

// ✅ 分块处理（利用事件循环）
async function processInChunks(arr, chunkSize = 100) {
  for (let i = 0; i < arr.length; i += chunkSize) {
    const chunk = arr.slice(i, i + chunkSize);
    chunk.forEach(processItem);
    
    // 让出主线程，允许 UI 更新和其他任务
    await new Promise(resolve => setTimeout(resolve, 0));
  }
}

// ✅ 使用 queueMicrotask（更轻量）
function processWithMicrotasks(arr, index = 0) {
  if (index >= arr.length) return;
  
  processItem(arr[index]);
  
  if ((index + 1) % 100 === 0) {
    // 每 100 个让出一次主线程
    setTimeout(() => processWithMicrotasks(arr, index + 1), 0);
  } else {
    queueMicrotask(() => processWithMicrotasks(arr, index + 1));
  }
}

// ✅ Web Worker（真正的多线程）

self.onmessage = (e) => {
  const result = heavyComputation(e.data);
```

## 五、综合实战题

### 题 1：实现一个限制并发数的异步任务调度器

```javascript
class Scheduler {
  constructor(maxConcurrent) {
    this.maxConcurrent = maxConcurrent;
    this.running = 0;
    this.queue = [];
  }
  
  add(taskFactory) {
    return new Promise((resolve, reject) => {
      this.queue.push({ taskFactory, resolve, reject });
      this._run();
    });
  }
  
  _run() {
    while (this.running < this.maxConcurrent && this.queue.length > 0) {
      const { taskFactory, resolve, reject } = this.queue.shift();
      this.running++;
      
      Promise.resolve(taskFactory())
        .then(resolve)
        .catch(reject)
        .finally(() => {
          this.running--;
          this._run();  // 触发下一个任务
        });
    }
  }
}

// 使用
const scheduler = new Scheduler(2);

const timeout = (t) => new Promise(r => setTimeout(r, t));

scheduler.add(() => timeout(100)).then(() => console.log('1 done'));
scheduler.add(() => timeout(200)).then(() => console.log('2 done'));
scheduler.add(() => timeout(100)).then(() => console.log('3 done'));
scheduler.add(() => timeout(300)).then(() => console.log('4 done'));
// 1 和 2 并行 → 1 先完成 → 3 开始 → 3 完成 → 4 开始
```

### 题 2：实现一个带缓存的异步函数

```javascript
function asyncMemoize(fn, keyFn = JSON.stringify) {
  const cache = new Map();
  
  return async function(...args) {
    const key = keyFn(args);
    
    if (cache.has(key)) {
      return cache.get(key);  // 返回已有的 Promise
    }
    
    const promise = fn.apply(this, args);
    cache.set(key, promise);
    
    // 完成后清理缓存（可选）
    promise.finally(() => cache.delete(key));
    
    return promise;
  };
}

// 使用：防止重复请求
const fetchUser = asyncMemoize(async (id) => {
  const res = await fetch(`/api/users/${id}`);
  return res.json();
});

// 同一时刻多次调用，只发一次请求
Promise.all([
  fetchUser(1),
  fetchUser(1),
  fetchUser(1)
]);  // 只发 1 个 HTTP 请求
```

### 题 3：实现一个事件发射器（EventEmitter）

```javascript
class EventEmitter {
  constructor() {
    this.events = new Map();
  }
  
  on(event, listener) {
    if (!this.events.has(event)) {
      this.events.set(event, []);
    }
    this.events.get(event).push(listener);
    return this;  // 链式调用
  }
  
  once(event, listener) {
    const onceWrapper = (...args) => {
      listener.apply(this, args);
      this.off(event, onceWrapper);
    };
    onceWrapper._original = listener;
    return this.on(event, onceWrapper);
  }
  
  off(event, listener) {
    const listeners = this.events.get(event);
    if (!listeners) return this;
    
    this.events.set(event, listeners.filter(
      l => l !== listener && l._original !== listener
    ));
    return this;
  }
  
  emit(event, ...args) {
    const listeners = this.events.get(event);
    if (!listeners) return false;
    
    // 复制数组防止监听器中修改数组
    listeners.slice().forEach(listener => listener.apply(this, args));
    return true;
  }
}

// 使用
const emitter = new EventEmitter();

emitter.on('data', (val) => console.log('data:', val));
emitter.once('once', (val) => console.log('once:', val));

emitter.emit('data', 1);   // data: 1
emitter.emit('data', 2);   // data: 2
emitter.emit('once', 3);   // once: 3
emitter.emit('once', 4);   // (无输出，已移除)
```

## 💡 核心收获

1. **闭包不是魔法** — 就是函数持有对外层词法环境的引用。V8 会做闭包裁剪优化。

6. **性能优化的核心** — 不要让同步代码阻塞主线程，用分块/Worker/RAF 等手段。
