## 重构后代码（符合 Airbnb 规范）

```javascript
/**
 * 计算含税价格
 * @param {number} basePrice - 基础价格
 * @param {number} quantity - 数量
 * @param {number} discount - 折扣金额
 * @returns {number} 含税总价
 */
function calculateTotalPrice(basePrice, quantity, discount = 0) {
  // 参数验证
  if (basePrice <= 0) {
    return 0;
  }

  // 计算基础总额
  const subtotal = quantity > 0 ? basePrice * quantity : basePrice;

  // 应用折扣
  const afterDiscount = subtotal - discount;

  // 计算税费 (10%)
  const TAX_RATE = 0.1;
  const tax = afterDiscount * TAX_RATE;

  // 返回含税总价
  return afterDiscount + tax;
}

// ============ 数组处理 ============

const data = [1, 2, 3, 4, 5];

// 使用 filter 和 map 替代 for 循环
const newData = data.filter((item) => item % 2 === 0).map((item) => item * 2);

console.log(newData); // [4, 8]

// ============ 字符串模板 ============

const obj = {
  name: "test",
  value: 123,
};

// 使用模板字符串替代字符串拼接
const str = `hello ${obj.name} your value is ${obj.value}`;

console.log(str);
```

## 重构要点总结

### 1. 变量声明

```javascript
// ❌ 错误
var count = 0;

// ✅ 正确
const count = 0; // 不变量用 const
let count = 0; // 变量用 let
```

### 2. 函数命名与格式

```javascript
// ❌ 错误
function calc(a, b, c) {}

// ✅ 正确
function calculateTotal(basePrice, quantity, discount) {
  // 函数体
}
```

### 3. 条件语句

```javascript
// ❌ 错误 - 嵌套过深
if (a > 0) {
  if (b > 0) {
    // ...
  }
}

// ✅ 正确 - 提前返回
if (a <= 0) return 0;
if (b <= 0) return a;
// ...
```

### 4. 数组操作

```javascript
// ❌ 错误 - 传统 for 循环
for (var i = 0; i < arr.length; i++) {
  if (arr[i] % 2 === 0) {
    result.push(arr[i] * 2);
  }
}

// ✅ 正确 - 函数式方法
const result = arr.filter((item) => item % 2 === 0).map((item) => item * 2);
```

### 5. 字符串处理

```javascript
// ❌ 错误 - 字符串拼接
var str = "hello " + name + " you are " + age + " years old";

// ✅ 正确 - 模板字符串
const str = `hello ${name} you are ${age} years old`;
```

### 6. 对象字面量

```javascript
// ❌ 错误
var obj = { name: "test", value: 123 };

// ✅ 正确
const obj = {
  name: "test",
  value: 123,
};
```

### 7. 魔法数字

```javascript
// ❌ 错误
result = result + result * 0.1;

// ✅ 正确
const TAX_RATE = 0.1;
result = result + result * TAX_RATE;
```

### 8. JSDoc 注释

```javascript
/**
 * 函数说明
 * @param {type} paramName - 参数说明
 * @returns {type} 返回值说明
 */
function myFunction(paramName) {
  // ...
}
```
