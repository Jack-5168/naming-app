## 一、关键渲染路径优化 (Critical Rendering Path)

### 1.1 理解 CRP

浏览器从 HTML/CSS/JS 到像素渲染的 6 个步骤：

```
HTML → DOM 树
CSS  → CSSOM 树
DOM + CSSOM → 渲染树 (Render Tree)
渲染树 → 布局 (Layout/Reflow)
布局 → 绘制 (Paint)
绘制 → 合成 (Composite)
```

**关键原则：**

- CSS 阻塞渲染（必须下载+解析完才能构建 CSSOM）
- JS 阻塞 DOM 解析（默认同步执行会暂停 HTML 解析）
- 首屏关键资源应内联或 preload

### 1.2 关键 CSS 内联 + 异步加载非关键 CSS

```javascript
/**
 * 关键 CSS 提取与异步加载策略
 */
class CriticalCSSLoader {
  constructor() {
    this.criticalCSS = "";
    this.nonCriticalURLs = [];
  }

  /**
   * 内联关键 CSS（首屏必需样式）
   * 实际项目中由 build 工具（如 critical）自动提取
   */
  inlineCritical(css) {
    const style = document.createElement("style");
    style.textContent = css;
    document.head.appendChild(style);
  }

  /**
   * 异步加载非关键 CSS
   * 利用 <link rel="preload"> + media="print" 技巧
   */
  loadNonCriticalCSS(url) {
    const link = document.createElement("link");
    link.rel = "preload";
    link.href = url;
    link.as = "style";
    link.onload = () => {
      link.rel = "stylesheet";
    };
    // 先设为 print（不阻塞渲染），加载完成后切换为 all
    link.media = "print";
    link.onload = function () {
      this.media = "all";
    };
    document.head.appendChild(link);
  }

  /**
   * 批量异步加载
   */
  loadAll(urls) {
    urls.forEach((url) => this.loadNonCriticalCSS(url));
  }
}
```

```javascript
/**
 * 高性能 DOM 操作工具
 */
class PerformantDOM {
  /**
   * 批量 DOM 操作 — DocumentFragment
   */
  static appendBatch(container, items, renderItem) {
    const fragment = document.createDocumentFragment();

    items.forEach((item) => {
      const el = renderItem(item);
      fragment.appendChild(el);
    });

    container.appendChild(fragment); // 只触发 1 次重排
  }

  /**
   * 离屏操作 — clone + replace
   */
  static offscreenUpdate(element, updateFn) {
    const clone = element.cloneNode(true);
    updateFn(clone);
    element.parentNode.replaceChild(clone, element);
  }

  /**
   * 使用 CSS transform 代替 layout 属性
   * transform 只触发合成，不触发重排
   */
  static animateMove(element, x, y) {
    // ❌ 触发重排
    // element.style.left = x + 'px';
    // element.style.top = y + 'px';

    // ✅ 只触发合成（GPU 加速）
    element.style.transform = `translate(${x}px, ${y}px)`;
  }

  /**
   * will-change 提示浏览器提前优化
   * ⚠️ 不要滥用，只在确需动画的元素上使用
   */
  static prepareAnimation(element, properties) {
    element.style.willChange = properties.join(", ");

    // 动画结束后移除 will-change
    element.addEventListener(
      "transitionend",
      () => {
        element.style.willChange = "auto";
      },
      { once: true },
    );
  }

  /**
   * contain 属性 — 隔离渲染区域
   */
  static isolateRendering(element) {
    // layout: 内部布局变化不影响外部
    // paint: 内部绘制变化不影响外部
    // size: 元素尺寸不依赖内部内容
    element.style.contain = "layout paint size";
  }
}

// ============ 使用示例 ============

  return li;
});
```

```javascript
/**
 * 智能请求管理器
 */
class RequestManager {
  constructor() {
    this.cache = new Map();
    this.pending = new Map();
    this.abortControllers = new Map();
  }

  /**
   * 请求去重 — 相同 URL 只发一次请求
   */
  async fetch(url, options = {}) {
    // 检查缓存
    if (this.cache.has(url) && !options.bypassCache) {
      return this.cache.get(url);
    }

    // 检查是否有进行中的相同请求
    if (this.pending.has(url)) {
      return this.pending.get(url);
    }

    // 创建 AbortController 用于取消
    const controller = new AbortController();
    const fetchOptions = { ...options, signal: controller.signal };

    const promise = fetch(url, fetchOptions)
      .then((res) => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return res.json();
      })
      .then((data) => {
        this.cache.set(url, data);
        this.pending.delete(url);
        this.abortControllers.delete(url);
        return data;
      })
      .catch((err) => {
        this.pending.delete(url);
        this.abortControllers.delete(url);
        if (err.name === "AbortError") return null;
        throw err;
      });

    this.pending.set(url, promise);
    this.abortControllers.set(url, controller);

    return promise;
  }

  /**
   * 取消进行中的请求
   */
  cancel(url) {
    const controller = this.abortControllers.get(url);
    if (controller) {
      controller.abort();
    }
  }

  /**
   * 批量请求 — 使用 Promise.all 并行
   */
  async fetchBatch(urls) {
    const results = await Promise.allSettled(
      urls.map((url) => this.fetch(url)),
    );

    return results.map((result, i) => ({
      url: urls[i],
      status: result.status,
      data: result.status === "fulfilled" ? result.value : result.reason,
    }));
  }

  /**
   * 请求队列 — 限制并发数
   */
  async fetchQueue(urls, maxConcurrency = 4) {
    const results = [];
    const executing = new Set();

    for (const url of urls) {
      const p = this.fetch(url).then((data) => {
        results.push({ url, data });
        executing.delete(p);
      });
      executing.add(p);

      if (executing.size >= maxConcurrency) {
        await Promise.race(executing);
      }
    }

    await Promise.all(executing);
    return results;
  }

  /**
   * 缓存清理
   */
  clearCache(url) {
    if (url) {
      this.cache.delete(url);
    } else {
      this.cache.clear();
    }
  }
}

// ============ 使用示例 ============
const api = new RequestManager();

// 请求去重：多次调用相同 URL 只发一次请求
const [users, posts] = await Promise.all([
  api.fetch("/api/users"), // 实际只发 1 次请求
  api.fetch("/api/users"), // 返回同一个 Promise
  api.fetch("/api/posts"),
]);

// 并发限制：最多 3 个并发请求
const results = await api.fetchQueue(
  [
    "/api/data/1",
    "/api/data/2",
    "/api/data/3",
    "/api/data/4",
    "/api/data/5",
    "/api/data/6",
  ],
  3,
);
```

### 3.2 Service Worker 缓存策略

```javascript
// sw.js — Service Worker 缓存策略
const CACHE_NAME = "app-v1";
const STATIC_ASSETS = ["/", "/css/main.css", "/js/app.js", "/images/logo.svg"];

// Install — 预缓存静态资源
self.addEventListener("install", (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(STATIC_ASSETS)),
  );
  self.skipWaiting();
});

// Activate — 清理旧缓存
self.addEventListener("activate", (event) => {
  event.waitUntil(
    caches
      .keys()
      .then((keys) =>
        Promise.all(
          keys
            .filter((key) => key !== CACHE_NAME)
            .map((key) => caches.delete(key)),
        ),
      ),
  );
  self.clients.claim();
});

// Fetch — 根据不同资源类型使用不同策略
self.addEventListener("fetch", (event) => {
  const { request } = event;
  const url = new URL(request.url);

  // API 请求：Network-First（网络优先，失败时回退缓存）
  if (url.pathname.startsWith("/api/")) {
    event.respondWith(networkFirst(request));
    return;
  }

  // 静态资源：Cache-First（缓存优先，缓存未命中时请求网络）
  if (/\.(css|js|png|jpg|svg|woff2)$/.test(url.pathname)) {
    event.respondWith(cacheFirst(request));
    return;
  }

  // HTML 页面：Stale-While-Revalidate
  if (request.headers.get("accept").includes("text/html")) {
    event.respondWith(staleWhileRevalidate(request));
    return;
  }
});

async function cacheFirst(request) {
  const cache = await caches.open(CACHE_NAME);
  const cached = await cache.match(request);
  if (cached) return cached;

  try {
    const response = await fetch(request);
    if (response.ok) {
      cache.put(request, response.clone());
    }
    return response;
  } catch (err) {
    return new Response("Offline", { status: 503 });
  }
}

async function networkFirst(request) {
  try {
    const response = await fetch(request);
    const cache = await caches.open(CACHE_NAME);
    cache.put(request, response.clone());
    return response;
  } catch (err) {
    const cache = await caches.open(CACHE_NAME);
    const cached = await cache.match(request);
    return cached || new Response("Offline", { status: 503 });
  }
}

async function staleWhileRevalidate(request) {
  const cache = await caches.open(CACHE_NAME);
  const cached = await cache.match(request);

  const fetchPromise = fetch(request)
    .then((response) => {
      if (response.ok) cache.put(request, response.clone());
      return response;
    })
    .catch(() => cached);

  return cached || fetchPromise;
}
```

```javascript
/**
 * Worker 管理器 — 简化 Web Worker 使用
 */
class WorkerPool {
  constructor(workerUrl, size = navigator.hardwareConcurrency || 4) {
    this.workers = Array.from({ length: size }, () => new Worker(workerUrl));
    this.queue = [];
    this.busy = new Set();
  }

  run(data) {
    return new Promise((resolve, reject) => {
      const task = { data, resolve, reject };

      const availableWorker = this.workers.find((w) => !this.busy.has(w));

      if (availableWorker) {
        this.execute(availableWorker, task);
      } else {
        this.queue.push(task);
      }
    });
  }

  execute(worker, task) {
    this.busy.add(worker);

    worker.onmessage = (e) => {
      task.resolve(e.data);
      this.busy.delete(worker);
      this.processQueue();
    };

    worker.onerror = (err) => {
      task.reject(err);
      this.busy.delete(worker);
      this.processQueue();
    };

    worker.postMessage(task.data);
  }

  processQueue() {
    if (this.queue.length > 0) {
      const availableWorker = this.workers.find((w) => !this.busy.has(w));
      if (availableWorker) {
        this.execute(availableWorker, this.queue.shift());
      }
    }
  }

  terminate() {
    this.workers.forEach((w) => w.terminate());
    this.workers = [];
    this.queue = [];
    this.busy.clear();
  }
}

// ============ 使用示例：大数据排序 ============
```

### 4.3 长任务拆分 (Long Task Splitting)

```javascript
/**
 * 任务调度器 — 将长任务拆分为多个短任务
 * 避免阻塞主线程，保持 60fps
 */
class TaskScheduler {
  constructor(options = {}) {
    this.yieldInterval = options.yieldInterval || 5; // 每 5ms 让出控制权
    this.tasks = [];
    this.running = false;
  }

  /**
   * 调度单个任务
   */
  scheduleTask(fn, priority = "normal") {
    return new Promise((resolve) => {
      this.tasks.push({ fn, resolve, priority, time: Date.now() });
      this.tasks.sort((a, b) => {
        const priorityOrder = { high: 0, normal: 1, low: 2 };
        return (
          priorityOrder[a.priority] - priorityOrder[b.priority] ||
          a.time - b.time
        );
      });
      if (!this.running) this.process();
    });
  }

  /**
   * 调度批量任务 — 使用 requestIdleCallback 或时间分片
   */
  async scheduleBatch(items, processFn, options = {}) {
    const { onProgress, chunkSize = 50, priority = "normal" } = options;

    const results = [];

    for (let i = 0; i < items.length; i += chunkSize) {
      const chunk = items.slice(i, i + chunkSize);

      // 使用 scheduleTask 让出控制权
      const chunkResults = await this.scheduleTask(() => {
        return chunk.map((item) => processFn(item));
      }, priority);

      results.push(...chunkResults);

      if (onProgress) {
        onProgress({
          progress: (i + chunk.length) / items.length,
          processed: i + chunk.length,
          total: items.length,
        });
      }
    }

    return results;
  }

  async process() {
    this.running = true;

    while (this.tasks.length > 0) {
      const task = this.tasks.shift();

      // 使用 scheduler.postTask（如果可用）或 requestAnimationFrame
      if ("scheduler" in window) {
        const priority =
          task.priority === "high"
            ? "user-blocking"
            : task.priority === "normal"
              ? "user-visible"
              : "background";

        await scheduler.postTask(
          async () => {
            const result = await task.fn();
            task.resolve(result);
          },
          { priority },
        );
      } else {
        // 降级：使用 MessageChannel 实现微任务调度
        await new Promise((resolve) => {
          const channel = new MessageChannel();
          channel.port1.onmessage = () => resolve();
          channel.port2.postMessage(null);
        }).then(async () => {
          const result = await task.fn();
          task.resolve(result);
        });
      }
    }

    this.running = false;
  }
}

// ============ 使用示例 ============

const scheduler = new TaskScheduler();

// 处理大量数据而不阻塞 UI
async function processData() {
  const largeArray = Array.from({ length: 100000 }, (_, i) => ({
    id: i,
    value: Math.random(),
  }));

  const results = await scheduler.scheduleBatch(
    largeArray,
    (item) => {
      // 模拟计算密集型操作
      return { ...item, processed: Math.sqrt(item.value) * 100 };
```

### 5.1 Core Web Vitals 指标

| 指标 | 含义         | 良好阈值 | 优化方向                       |
| ---- | ------------ | -------- | ------------------------------ |

| INP  | 交互到下一帧 | < 200ms  | 减少长任务、代码分割           |

### 5.2 性能监控面板

```javascript
/**
 * Web Vitals 监控器
 */
class PerformanceMonitor {
  constructor(options = {}) {
    this.onLCP = options.onLCP || console.log;
    this.onINP = options.onINP || console.log;
    this.onCLS = options.onCLS || console.log;
    this.onFCP = options.onFCP || console.log;
    this.onTTFB = options.onTTFB || console.log;
    this.metrics = {};
  }

  init() {
    this.observeLCP();
    this.observeINP();
    this.observeCLS();
    this.observeFCP();
    this.observeTTFB();
  }

  /**
   * LCP — 最大内容绘制
   */
  observeLCP() {
    new PerformanceObserver((list) => {
      const entries = list.getEntries();
      const lastEntry = entries[entries.length - 1];

      this.metrics.lcp = lastEntry.startTime;
      this.onLCP({
        value: lastEntry.startTime,
        rating:
          lastEntry.startTime < 2500
            ? "good"
            : lastEntry.startTime < 4000
              ? "needs-improvement"
              : "poor",
        element: lastEntry.element?.tagName,
      });
    }).observe({ type: "largest-contentful-paint", buffered: true });
  }

  /**
   * INP — 交互到下一帧（替代 FID）
   */
  observeINP() {
    const interactions = new Map();
    let maxINP = 0;

    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        const interactionId = entry.interactionId || entry.name;
        const existing = interactions.get(interactionId);

        if (!existing || entry.processingStart > existing.processingStart) {
          interactions.set(interactionId, entry);
        }
      }

      for (const [id, entry] of interactions) {
        const inp = entry.processingStart - entry.startTime;
        if (inp > maxINP) maxINP = inp;
      }

      this.metrics.inp = maxINP;
      this.onINP({
        value: maxINP,
        rating:
          maxINP < 200 ? "good" : maxINP < 500 ? "needs-improvement" : "poor",
      });
    }).observe({ type: "event", buffered: true, durationThreshold: 0 });
  }

  /**
   * CLS — 累积布局偏移
   */
  observeCLS() {
    let clsValue = 0;
    let sessionValue = 0;
    let sessionEntries = [];

    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (!entry.hadRecentInput) {
          const firstEntryTime = sessionEntries[0]?.startTime;
          const lastEntryTime =
            sessionEntries[sessionEntries.length - 1]?.startTime;

          if (
            sessionValue &&
            entry.startTime - lastEntryTime < 1000 &&
            entry.startTime - firstEntryTime < 5000
          ) {
            sessionValue += entry.value;
            sessionEntries.push(entry);
          } else {
            sessionValue = entry.value;
            sessionEntries = [entry];
          }
        }
      }

      clsValue = sessionValue;
      this.metrics.cls = clsValue;
      this.onCLS({
        value: clsValue,
        rating:
          clsValue < 0.1
            ? "good"
            : clsValue < 0.25
              ? "needs-improvement"
              : "poor",
      });
    }).observe({ type: "layout-shift", buffered: true });
  }

  /**
   * FCP — 首次内容绘制
   */
  observeFCP() {
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.name === "first-contentful-paint") {
          this.metrics.fcp = entry.startTime;
          this.onFCP({
            value: entry.startTime,
            rating: entry.startTime < 1800 ? "good" : "needs-improvement",
          });
        }
      }
    }).observe({ type: "paint", buffered: true });
  }

  /**
   * TTFB — 首字节时间
   */
  observeTTFB() {
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.entryType === "navigation") {
          this.metrics.ttfb = entry.responseStart;
          this.onTTFB({
            value: entry.responseStart,
            rating: entry.responseStart < 800 ? "good" : "needs-improvement",
          });
        }
      }
    }).observe({ type: "navigation", buffered: true });
  }

  /**
   * 获取所有指标
   */
  getMetrics() {
    return { ...this.metrics };
  }

  /**
   * 生成性能报告
   */
  getReport() {
    const m = this.metrics;
    const report = [];

    if (m.lcp)
      report.push(`LCP: ${m.lcp.toFixed(0)}ms ${m.lcp < 2500 ? "✅" : "⚠️"}`);
    if (m.inp)
      report.push(`INP: ${m.inp.toFixed(0)}ms ${m.inp < 200 ? "✅" : "⚠️"}`);
    if (m.cls)
      report.push(`CLS: ${m.cls.toFixed(3)} ${m.cls < 0.1 ? "✅" : "⚠️"}`);
    if (m.fcp)
      report.push(`FCP: ${m.fcp.toFixed(0)}ms ${m.fcp < 1800 ? "✅" : "⚠️"}`);
    if (m.ttfb)
      report.push(`TTFB: ${m.ttfb.toFixed(0)}ms ${m.ttfb < 800 ? "✅" : "⚠️"}`);

    return report.join("\n");
  }
}

// ============ 使用示例 ============
const monitor = new PerformanceMonitor({
  onLCP: (m) => console.log(`📊 LCP: ${m.value.toFixed(0)}ms (${m.rating})`),
  onINP: (m) => console.log(`📊 INP: ${m.value.toFixed(0)}ms (${m.rating})`),
  onCLS: (m) => console.log(`📊 CLS: ${m.value.toFixed(3)} (${m.rating})`),
});

monitor.init();

// 页面加载完成后输出报告
window.addEventListener("load", () => {
  setTimeout(() => {
    console.log("\n=== 性能报告 ===");
    console.log(monitor.getReport());
  }, 3000);
});
```

### 6.2 优化后

```javascript
// ✅ 全面优化版本
class OptimizedECommerceApp {
  constructor() {
    this.products = [];
    this.page = 0;
    this.pageSize = 24;
    this.hasMore = true;
    this.searchQuery = "";
    this.resources = { timers: [], observers: [], listeners: [] };
    this.init();
  }

  init() {
    // 优化 1: 分页加载 + Intersection Observer 无限滚动
    this.loadProducts();
    this.initInfiniteScroll();

    // 优化 2: 防抖搜索
    const searchInput = document.getElementById("search");
    const debouncedSearch = this.debounce((query) => {
      this.searchQuery = query;
      this.page = 0;
      this.products = [];
      this.hasMore = true;
      this.loadProducts();
    }, 300);

    searchInput.addEventListener("input", (e) => {
      debouncedSearch(e.target.value);
    });
    this.resources.listeners.push({ target: searchInput, event: "input" });

    // 优化 3: 节流滚动
    const throttledScroll = this.throttle(() => {
      this.checkLoadMore();
      this.updateScrollIndicator();
    }, 100);

    window.addEventListener("scroll", throttledScroll);
    this.resources.listeners.push({
      target: window,
      event: "scroll",
      handler: throttledScroll,
    });

    // 优化 4: 图片懒加载
    this.initImageObserver();

    // 优化 5: 定时器可清理
    const priceTimer = setInterval(() => this.refreshPrices(), 30000);
    this.resources.timers.push(priceTimer);
  }

  async loadProducts() {
    if (!this.hasMore) return;

    const offset = this.page * this.pageSize;
    const url = `/api/products?limit=${this.pageSize}&offset=${offset}&q=${encodeURIComponent(this.searchQuery)}`;

    const response = await fetch(url);
    const data = await response.json();

    if (data.length < this.pageSize) this.hasMore = false;

    this.products.push(...data);
    this.renderProducts(data);
    this.page++;
  }

  renderProducts(products) {
    const container = document.getElementById("products");
    const fragment = document.createDocumentFragment();

    products.forEach((p) => {
      const card = document.createElement("div");
      card.className = "product-card";
      card.innerHTML = `
        <img data-src="${p.image}" alt="${p.name}" 
             style="aspect-ratio: 1; width: 100%;">
        <h3>${p.name}</h3>
        <p>¥${p.price}</p>
      `;
      fragment.appendChild(card);
    });

    container.appendChild(fragment); // 只触发 1 次重排

    // 为新添加的图片注册懒加载
    container.querySelectorAll("img[data-src]").forEach((img) => {
      this.imageObserver.observe(img);
    });
  }

  initInfiniteScroll() {
    const sentinel = document.getElementById("scroll-sentinel");
    this.scrollObserver = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting && this.hasMore) {
          this.loadProducts();
        }
      },
      { rootMargin: "400px" },
    );

    this.scrollObserver.observe(sentinel);
    this.resources.observers.push(this.scrollObserver);
  }

  initImageObserver() {
    this.imageObserver = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            const img = entry.target;
            img.src = img.dataset.src;
            this.imageObserver.unobserve(img);
          }
        });
      },
      { rootMargin: "200px" },
    );

    this.resources.observers.push(this.imageObserver);
  }

  checkLoadMore() {
    const scrollBottom = window.scrollY + window.innerHeight;
    const docHeight = document.documentElement.scrollHeight;
    // 由 Intersection Observer 处理，这里只更新 UI
  }

  updateScrollIndicator() {
    const scrollTop = window.scrollY;
    const docHeight =
      document.documentElement.scrollHeight - window.innerHeight;
    const progress = docHeight > 0 ? (scrollTop / docHeight) * 100 : 0;

    const bar = document.getElementById("scroll-progress");
    if (bar) bar.style.width = `${progress}%`;
  }

  refreshPrices() {
    // 静默更新价格，不影响用户操作
    this.products.forEach((p) => {
      const el = document.querySelector(`[data-product-id="${p.id}"] .price`);
      if (el) el.textContent = `¥${p.price}`;
    });
  }

  // 工具方法
  debounce(fn, delay) {
    let timer;
    return (...args) => {
      clearTimeout(timer);
      timer = setTimeout(() => fn.apply(this, args), delay);
    };
  }

  throttle(fn, limit) {
    let last = 0;
    return (...args) => {
      const now = Date.now();
      if (now - last >= limit) {
        last = now;
        fn.apply(this, args);
      }
    };
  }

  // 清理所有资源
  destroy() {
    this.resources.timers.forEach((id) => clearInterval(id));
    this.resources.observers.forEach((obs) => obs.disconnect());
    this.resources.listeners.forEach(({ target, event, handler }) => {
      if (handler) target.removeEventListener(event, handler);
    });
    this.products = null;
  }
}
```
