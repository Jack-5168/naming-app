## 模式 1：观察者模式 (Observer Pattern)

### 概念理解
观察者模式定义了一种一对多的依赖关系，当一个对象的状态发生改变时，所有依赖于它的对象都得到通知并被自动更新。

### 应用场景
- 事件处理系统
- MVVM 框架的数据绑定
- 发布/订阅系统
- 响应式编程

### 实现代码

```javascript
/**
 * 被观察者类 (Subject)
 */
class Subject {
    constructor() {
        this.observers = [];
        this.state = null;
    }
    
    // 添加观察者
    addObserver(observer) {
        if (!this.observers.includes(observer)) {
            this.observers.push(observer);
        }
        return this;
    }
    
    // 移除观察者
    removeObserver(observer) {
        this.observers = this.observers.filter(obs => obs !== observer);
        return this;
    }
    
    // 通知所有观察者
    notify() {
        this.observers.forEach(observer => observer.update(this.state));
        return this;
    }
    
    // 设置状态并通知
    setState(state) {
        this.state = state;
        this.notify();
        return this;
    }
    
    getState() {
        return this.state;
    }
}

/**
 * 观察者类 (Observer)
 */
class Observer {
    constructor(name) {
        this.name = name;
    }
    
    update(state) {
        console.log(`[观察者:${this.name}] 收到通知，新状态: ${state}`);
    }
}

// ============ 使用示例 ============

// 创建被观察者
const weatherStation = new Subject();

// 创建观察者
const phoneDisplay = new Observer('手机显示');
const tabletDisplay = new Observer('平板显示');
const weatherApp = new Observer('天气 App');

// 添加观察者
weatherStation
    .addObserver(phoneDisplay)
    .addObserver(tabletDisplay)
    .addObserver(weatherApp);

// 改变状态，自动通知所有观察者
console.log('=== 温度更新为 25°C ===');
weatherStation.setState('25°C');

console.log('\n=== 温度更新为 28°C ===');
weatherStation.setState('28°C');

// 移除一个观察者
console.log('\n=== 移除平板显示 ===');
weatherStation.removeObserver(tabletDisplay);

console.log('\n=== 温度更新为 30°C ===');
weatherStation.setState('30°C');
```

### 进阶：事件发射器 (Event Emitter)

```javascript
class EventEmitter {
    constructor() {
        this.events = {};
    }
    
    // 订阅事件
    on(event, listener) {
        if (!this.events[event]) {
            this.events[event] = [];
        }
        this.events[event].push(listener);
        return this;
    }
    
    // 一次性订阅
    once(event, listener) {
        const onceListener = (...args) => {
            listener(...args);
            this.off(event, onceListener);
        };
        return this.on(event, onceListener);
    }
    
    // 取消订阅
    off(event, listener) {
        if (!this.events[event]) return this;
        this.events[event] = this.events[event].filter(l => l !== listener);
        if (this.events[event].length === 0) {
            delete this.events[event];
        }
        return this;
    }
    
    // 触发事件
    emit(event, ...args) {
        if (!this.events[event]) return false;
        this.events[event].forEach(listener => listener(...args));
        return true;
    }
    
    // 获取监听器数量
    listenerCount(event) {
        return this.events[event] ? this.events[event].length : 0;
    }
}

// ============ 使用示例 ============

const emitter = new EventEmitter();

// 订阅事件
emitter.on('data', (data) => {
    console.log('收到数据:', data);
});

emitter.on('error', (err) => {
    console.error('发生错误:', err);
});

// 一次性订阅
emitter.once('ready', () => {
    console.log('准备就绪（只执行一次）');
});

// 触发事件
emitter.emit('data', { id: 1, name: '测试' });
emitter.emit('ready');
emitter.emit('ready'); // 这次不会执行
emitter.emit('error', new Error('测试错误'));
```

## 模式 2：策略模式 (Strategy Pattern)

### 概念理解
策略模式定义了一系列算法，并将每个算法封装起来，使它们可以相互替换。策略模式让算法的变化独立于使用算法的客户。

### 应用场景
- 多种排序算法切换
- 不同的支付方式
- 不同的验证策略
- 不同的导出格式

### 实现代码

```javascript
/**
 * 策略接口 (抽象)
 */
class SortStrategy {
    sort(arr) {
        throw new Error('必须实现 sort 方法');
    }
}

/**
 * 具体策略：冒泡排序
 */
class BubbleSort extends SortStrategy {
    sort(arr) {
        const result = [...arr];
        const n = result.length;
        for (let i = 0; i < n - 1; i++) {
            for (let j = 0; j < n - i - 1; j++) {
                if (result[j] > result[j + 1]) {
                    [result[j], result[j + 1]] = [result[j + 1], result[j]];
                }
            }
        }
        console.log('使用冒泡排序');
        return result;
    }
}

/**
 * 具体策略：快速排序
 */
class QuickSort extends SortStrategy {
    sort(arr) {
        const result = [...arr];
        const quickSort = (arr, left, right) => {
            if (left < right) {
                const pivotIndex = partition(arr, left, right);
                quickSort(arr, left, pivotIndex - 1);
                quickSort(arr, pivotIndex + 1, right);
            }
        };
        const partition = (arr, left, right) => {
            const pivot = arr[right];
            let i = left - 1;
            for (let j = left; j < right; j++) {
                if (arr[j] <= pivot) {
                    i++;
                    [arr[i], arr[j]] = [arr[j], arr[i]];
                }
            }
            [arr[i + 1], arr[right]] = [arr[right], arr[i + 1]];
            return i + 1;
        };
        quickSort(result, 0, result.length - 1);
        console.log('使用快速排序');
        return result;
    }
}

/**
 * 具体策略：归并排序
 */
class MergeSort extends SortStrategy {
    sort(arr) {
        const result = [...arr];
        const merge = (left, right) => {
            const res = [];
            let i = 0, j = 0;
            while (i < left.length && j < right.length) {
                if (left[i] <= right[j]) {
                    res.push(left[i++]);
                } else {
                    res.push(right[j++]);
                }
            }
            return [...res, ...left.slice(i), ...right.slice(j)];
        };
        const mergeSort = (arr) => {
            if (arr.length <= 1) return arr;
            const mid = Math.floor(arr.length / 2);
            const left = mergeSort(arr.slice(0, mid));
            const right = mergeSort(arr.slice(mid));
            return merge(left, right);
        };
        console.log('使用归并排序');
        return mergeSort(result);
    }
}

/**
 * 上下文类 (Context)
 */
class Sorter {
    constructor(strategy = new QuickSort()) {
        this.strategy = strategy;
    }
    
    // 动态切换策略
    setStrategy(strategy) {
        this.strategy = strategy;
        return this;
    }
    
    // 执行排序
    sort(arr) {
        return this.strategy.sort(arr);
    }
}

// ============ 使用示例 ============

const data = [64, 34, 25, 12, 22, 11, 90];

// 使用快速排序（默认）
const sorter = new Sorter();
console.log('\n=== 快速排序 ===');
console.log(sorter.sort(data));

// 切换为冒泡排序
sorter.setStrategy(new BubbleSort());
console.log('\n=== 冒泡排序 ===');
console.log(sorter.sort(data));

// 切换为归并排序
sorter.setStrategy(new MergeSort());
console.log('\n=== 归并排序 ===');
console.log(sorter.sort(data));
```

### 实战应用：支付策略

```javascript
// 支付策略接口
class PaymentStrategy {
    pay(amount) {
        throw new Error('必须实现 pay 方法');
    }
}

// 支付宝支付
class Alipay extends PaymentStrategy {
    constructor(username) {
        super();
        this.username = username;
    }
    
    pay(amount) {
        console.log(`[支付宝] ${this.username} 支付 ¥${amount.toFixed(2)}`);
        return { success: true, method: 'alipay' };
    }
}

// 微信支付
class WechatPay extends PaymentStrategy {
    constructor(openid) {
        super();
        this.openid = openid;
    }
    
    pay(amount) {
        console.log(`[微信支付] 用户${this.openid} 支付 ¥${amount.toFixed(2)}`);
        return { success: true, method: 'wechat' };
    }
}

// 信用卡支付
class CreditCard extends PaymentStrategy {
    constructor(cardNumber) {
        super();
        this.cardNumber = cardNumber;
    }
    
    pay(amount) {
        const last4 = this.cardNumber.slice(-4);
        console.log(`[信用卡] 卡号尾号${last4} 支付 ¥${amount.toFixed(2)}`);
        return { success: true, method: 'credit' };
    }
}

// 购物车上下文
class ShoppingCart {
    constructor() {
        this.items = [];
        this.paymentStrategy = null;
    }
    
    addItem(item) {
        this.items.push(item);
        return this;
    }
    
    getTotal() {
        return this.items.reduce((sum, item) => sum + item.price, 0);
    }
    
    setPayment(strategy) {
        this.paymentStrategy = strategy;
        return this;
    }
    
    checkout() {
        const total = this.getTotal();
        console.log(`\n订单总额：¥${total.toFixed(2)}`);
        if (!this.paymentStrategy) {
            throw new Error('请选择支付方式');
        }
        return this.paymentStrategy.pay(total);
    }
}

// ============ 使用示例 ============

const cart = new ShoppingCart();
cart
    .addItem({ name: '商品 A', price: 99 })
    .addItem({ name: '商品 B', price: 199 })
    .addItem({ name: '商品 C', price: 59});

// 使用支付宝
cart.setPayment(new Alipay('user123'));
cart.checkout();

// 使用微信支付
cart.setPayment(new WechatPay('wx_openid_456'));
cart.checkout();

// 使用信用卡
cart.setPayment(new CreditCard('1234567890123456'));
cart.checkout();
```
