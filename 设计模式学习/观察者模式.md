# 观察者模式 (Observer Pattern)

## 概念

观察者模式定义了一种一对多的依赖关系，当一个对象（主题）的状态发生变化时，所有依赖于它的对象（观察者）都会收到通知并自动更新。

## 适用场景

- 事件处理系统（DOM 事件、自定义事件）
- 发布 - 订阅系统（消息队列、事件总线）
- MVC 架构中的模型 - 视图同步
- 响应式编程（RxJS、Vue 响应式）
- 状态变化通知（Redux、Vuex）
- 实时数据更新（股票价格、聊天消息）

## 实现方式

### 1. 基础观察者模式

```javascript
// observer-basic.js
class Subject {
  constructor() {
    this.observers = [];
  }

  subscribe(observer) {
    if (!this.observers.includes(observer)) {
      this.observers.push(observer);
    }
    return () => this.unsubscribe(observer);
  }

  unsubscribe(observer) {
    this.observers = this.observers.filter((obs) => obs !== observer);
  }

  notify(data) {
    this.observers.forEach((observer) => observer.update(data));
  }

  getObserverCount() {
    return this.observers.length;
  }
}

class Observer {
  constructor(name) {
    this.name = name;
  }

  update(data) {
    console.log(`${this.name} received:`, data);
  }
}

// 使用示例
const subject = new Subject();

const obs1 = new Observer("Observer 1");
const obs2 = new Observer("Observer 2");
const obs3 = new Observer("Observer 3");

subject.subscribe(obs1);
subject.subscribe(obs2);
subject.subscribe(obs3);

subject.notify("Hello Observers!");
// Observer 1 received: Hello Observers!
// Observer 2 received: Hello Observers!
// Observer 3 received: Hello Observers!

const unsubscribe = subject.subscribe(new Observer("Temporary"));
subject.notify("Temporary observer active");
unsubscribe(); // 取消订阅
subject.notify("Temporary observer gone");
```

### 2. 事件总线（Event Bus）

```javascript
// observer-event-bus.js
class EventBus {
  constructor() {
    this.events = {};
  }

  on(event, callback) {
    if (!this.events[event]) {
      this.events[event] = [];
    }
    this.events[event].push(callback);
    return () => this.off(event, callback);
  }

  off(event, callback) {
    if (!this.events[event]) return;
    this.events[event] = this.events[event].filter((cb) => cb !== callback);
    if (this.events[event].length === 0) {
      delete this.events[event];
    }
  }

  emit(event, data) {
    if (!this.events[event]) return;
    this.events[event].forEach((callback) => callback(data));
  }

  once(event, callback) {
    const wrapper = (data) => {
      callback(data);
      this.off(event, wrapper);
    };
    return this.on(event, wrapper);
  }

  getEventNames() {
    return Object.keys(this.events);
  }

  getListenerCount(event) {
    return this.events[event]?.length || 0;
  }
}

// 使用示例
const bus = new EventBus();

// 订阅事件
bus.on("user:login", (user) => {
  console.log(`Welcome, ${user.name}!`);
});

bus.on("user:logout", (user) => {
  console.log(`Goodbye, ${user.name}!`);
});

bus.on("data:update", (data) => {
  console.log("Data updated:", data);
});

// 触发事件
bus.emit("user:login", { name: "Alice", id: 1 });
bus.emit("data:update", { field: "email", value: "alice@example.com" });
bus.emit("user:logout", { name: "Alice", id: 1 });

// 一次性事件
bus.once("notification", (msg) => {
  console.log("One-time notification:", msg);
});
bus.emit("notification", "First message"); // 会触发
bus.emit("notification", "Second message"); // 不会触发
```

### 3. 带优先级的观察者

```javascript
// observer-priority.js
class PrioritySubject {
  constructor() {
    this.observers = [];
  }

  subscribe(observer, priority = 0) {
    const entry = { observer, priority };
    this.observers.push(entry);
    this.observers.sort((a, b) => b.priority - a.priority);
    return () => this.unsubscribe(observer);
  }

  unsubscribe(observer) {
    this.observers = this.observers.filter(
      (entry) => entry.observer !== observer,
    );
  }

  notify(data) {
    console.log(`\nNotifying ${this.observers.length} observers...`);
    this.observers.forEach(({ observer, priority }) => {
      console.log(`  [Priority ${priority}]`);
      observer.update(data);
    });
  }
}

class PriorityObserver {
  constructor(name) {
    this.name = name;
  }

  update(data) {
    console.log(`    ${this.name} processed:`, data);
  }
}

// 使用示例
const subject = new PrioritySubject();

const high = new PriorityObserver("High Priority");
const medium = new PriorityObserver("Medium Priority");
const low = new PriorityObserver("Low Priority");

subject.subscribe(low, 1);
subject.subscribe(high, 10);
subject.subscribe(medium, 5);

subject.notify("Important update!");
// 输出顺序：High -> Medium -> Low
```

### 4. 异步观察者

```javascript
// observer-async.js
class AsyncSubject {
  constructor() {
    this.observers = [];
  }

  subscribe(observer) {
    this.observers.push(observer);
    return () => this.unsubscribe(observer);
  }

  unsubscribe(observer) {
    this.observers = this.observers.filter((obs) => obs !== observer);
  }

  async notify(data) {
    const results = await Promise.all(
      this.observers.map(async (observer) => {
        try {
          return await observer.update(data);
        } catch (error) {
          console.error("Observer error:", error);
          return { error: error.message };
        }
      }),
    );
    return results;
  }
}

class AsyncObserver {
  constructor(name, delay = 0) {
    this.name = name;
    this.delay = delay;
  }

  async update(data) {
    await new Promise((resolve) => setTimeout(resolve, this.delay));
    console.log(`${this.name} processed after ${this.delay}ms:`, data);
    return { observer: this.name, data, timestamp: Date.now() };
  }
}

// 使用示例
async function test() {
  const subject = new AsyncSubject();

  subject.subscribe(new AsyncObserver("Fast", 100));
  subject.subscribe(new AsyncObserver("Medium", 300));
  subject.subscribe(new AsyncObserver("Slow", 500));

  console.log("Starting async notification...");
  const results = await subject.notify("Async data");
  console.log("\nAll observers completed:", results);
}

test();
```

## 完整示例：新闻发布系统

```javascript
// observer-news-system.js
class NewsAgency {
  constructor() {
    this.channels = {};
    this.news = [];
  }

  registerChannel(channelName, observer) {
    if (!this.channels[channelName]) {
      this.channels[channelName] = [];
    }
    this.channels[channelName].push(observer);
    console.log(`${observer.name} subscribed to ${channelName}`);
    return () => this.unregisterChannel(channelName, observer);
  }

  unregisterChannel(channelName, observer) {
    if (this.channels[channelName]) {
      this.channels[channelName] = this.channels[channelName].filter(
        (obs) => obs !== observer,
      );
      console.log(`${observer.name} unsubscribed from ${channelName}`);
    }
  }

  publishNews(category, headline, content) {
    const news = {
      id: Date.now(),
      category,
      headline,
      content,
      timestamp: new Date().toISOString(),
    };

    this.news.push(news);
    console.log(`\n📰 BREAKING NEWS: ${headline}`);

    // 通知订阅该类别的观察者
    if (this.channels[category]) {
      this.channels[category].forEach((observer) => {
        observer.receiveNews(news);
      });
    }

    // 也通知订阅"all"的观察者
    if (this.channels["all"]) {
      this.channels["all"].forEach((observer) => {
        if (!this.channels[category]?.includes(observer)) {
          observer.receiveNews(news);
        }
      });
    }

    return news;
  }

  getNewsHistory(limit = 10) {
    return this.news.slice(-limit);
  }
}

class NewsChannel {
  constructor(name, platform) {
    this.name = name;
    this.platform = platform;
    this.receivedNews = [];
  }

  receiveNews(news) {
    this.receivedNews.push(news);
    console.log(`  [${this.platform}] ${this.name}: "${news.headline}"`);
  }

  getNewsCount() {
    return this.receivedNews.length;
  }

  getLatestNews() {
    return this.receivedNews[this.receivedNews.length - 1];
  }
}

// 使用示例
const agency = new NewsAgency();

// 创建新闻频道
const tv = new NewsChannel("CCTV", "TV");
const website = new NewsChannel("Sina", "Web");
const app = new NewsChannel("Toutiao", "Mobile");
const allNews = new NewsChannel("Google News", "Aggregator");

// 订阅不同类别
agency.registerChannel("politics", tv);
agency.registerChannel("politics", website);
agency.registerChannel("technology", app);
agency.registerChannel("technology", website);
agency.registerChannel("sports", tv);
agency.registerChannel("all", allNews);

// 发布新闻
agency.publishNews(
  "politics",
  "New Policy Announced",
  "Government announces new economic policy...",
);
agency.publishNews(
  "technology",
  "AI Breakthrough",
  "Scientists achieve major AI milestone...",
);
agency.publishNews(
  "sports",
  "Championship Results",
  "Team wins national championship...",
);

console.log("\n📊 Statistics:");
console.log(`${tv.name} received ${tv.getNewsCount()} news`);
console.log(`${website.name} received ${website.getNewsCount()} news`);
console.log(`${app.name} received ${app.getNewsCount()} news`);
console.log(`${allNews.name} received ${allNews.getNewsCount()} news`);
```

## 完整示例：响应式数据绑定

```javascript
// observer-reactive.js
class ReactiveData {
  constructor(initialData = {}) {
    this.data = initialData;
    this.observers = {};

    // 为每个属性创建 getter/setter
    Object.keys(initialData).forEach((key) => {
      this.observers[key] = [];
    });
  }

  subscribe(key, callback) {
    if (!this.observers[key]) {
      this.observers[key] = [];
    }
    this.observers[key].push(callback);
    return () => this.unsubscribe(key, callback);
  }

  unsubscribe(key, callback) {
    if (this.observers[key]) {
      this.observers[key] = this.observers[key].filter((cb) => cb !== callback);
    }
  }

  get(key) {
    return this.data[key];
  }

  set(key, value) {
    const oldValue = this.data[key];
    this.data[key] = value;

    if (this.observers[key]) {
      console.log(`"${key}" changed: ${oldValue} → ${value}`);
      this.observers[key].forEach((callback) => {
        callback(value, oldValue, key);
      });
    }
  }

  getAll() {
    return { ...this.data };
  }
}

// 使用示例
const state = new ReactiveData({
  count: 0,
  user: "Guest",
  theme: "light",
});

// 订阅变化
state.subscribe("count", (newVal, oldVal) => {
  console.log(`  UI Update: Count display changed to ${newVal}`);
});

state.subscribe("user", (newVal) => {
  console.log(`  UI Update: Welcome message changed to "${newVal}"`);
});

state.subscribe("theme", (newVal) => {
  console.log(`  UI Update: Theme switched to ${newVal} mode`);
});

console.log("Initial state:", state.getAll());

// 触发更新
state.set("count", 1);
state.set("count", 5);
state.set("user", "Alice");
state.set("theme", "dark");
```

## 优缺点

### 优点

- 解耦主题和观察者
- 支持广播通信
- 符合开闭原则（容易添加新观察者）
- 支持触发连锁反应

### 缺点

- 可能导致循环依赖
- 观察者过多时性能问题
- 观察者不知道变化来源
- 内存泄漏风险（未正确取消订阅）

## 注意事项

1. **内存泄漏**: 务必在适当时机取消订阅
2. **执行顺序**: 不要依赖观察者的执行顺序
3. **异常处理**: 单个观察者错误不应影响其他观察者
4. **性能**: 大量观察者时考虑批量通知或异步通知
5. **循环依赖**: 避免观察者和主题之间的循环引用
